function __fade() {
  if (__fadeEl) return __fadeEl;
  var host = document.querySelector("#pane3d") || document.body;
  __fadeEl = document.createElement("div");
  __fadeEl.id = "modeFade";
  host.appendChild(__fadeEl);
  return __fadeEl;
}
function __walkSwitch(work) {
  try {
    var f = __fade();
    f.style.transitionDuration = "120ms";
    f.style.opacity = .95;
    setTimeout(function () {
      try { work(); } catch (e) { }
      requestAnimationFrame(function () {
        requestAnimationFrame(function () {
          f.style.transitionDuration = "280ms";
          f.style.opacity = 0;
        });
      });
    }, 140);
  } catch (e) {
    try { work(); } catch (e2) { }
  }
}
